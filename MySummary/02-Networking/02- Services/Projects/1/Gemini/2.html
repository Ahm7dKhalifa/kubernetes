2 The "Port-to-YAML" Mapping Diagram

Step 1: NodePort

YAML: nodePort: 30001. The traffic enters the physical server here.

Step 2: Service Port

YAML: port: 80. The internal virtual IP of the service redirects traffic.

Step 3: Target Port

YAML: targetPort: 8080. Traffic finally reaches the application inside the container.

3. Full Application Manifests

apiVersion: v1
kind: Service
metadata:
  name: frontend-svc
spec:
  type: NodePort
  ports:
  - port: 80           # Port reachable inside cluster
    targetPort: 8080   # Port application listens on
    nodePort: 30001    # Port reachable from outside
  selector:
    app: frontend
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: frontend
spec:
  replicas: 2
  selector:
    matchLabels:
      app: frontend
  template:
    metadata:
      labels:
        app: frontend
    spec:
      containers:
      - name: nginx-container
        image: nginx:1.23
        ports:
        - containerPort: 8080
apiVersion: v1
kind: Service
metadata:
  name: backend-svc
spec:
  type: ClusterIP      # Default internal-only type
  ports:
  - port: 9000
    targetPort: 5000
  selector:
    app: backend
---
apiVersion: apps/v1
kind: Deployment
metadata:
  name: backend
spec:
  replicas: 5
  selector:
    matchLabels:
      app: backend
  template:
    metadata:
      labels:
        app: backend
    spec:
      containers:
      - name: python-api
        image: python:3.9-slim
        ports:
        - containerPort: 5000
apiVersion: v1
kind: Service
metadata:
  name: postgres-headless
spec:
  clusterIP: None      # Headless for StatefulSet direct access
  ports:
  - port: 5432
  selector:
    app: postgres
---
apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: postgres
spec:
  serviceName: "postgres-headless"
  replicas: 3
  template:
    metadata:
      labels:
        app: postgres
    spec:
      containers:
      - name: postgres-db
        image: postgres:15
