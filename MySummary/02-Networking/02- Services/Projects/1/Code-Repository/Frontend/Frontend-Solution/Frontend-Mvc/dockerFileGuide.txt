# Build the image
docker build -t frontend-mvc .

# Run the container
docker run -d -p 5000:8080 --name frontend-mvc frontend-mvc

# Access the frontend
http://localhost:5000

# Backend Service URL configuration
# Edit appsettings.json to configure: "BackendServiceUrl"
# For Docker: http://host.docker.internal:5000/WeatherForecast
# For K8s: http://backend-api-service:8080/WeatherForecast

# Publish to a registry (requires login first)
docker tag frontend-mvc yourregistry/frontend-mvc:latest
docker push yourregistry/frontend-mvc:latest

# Example with version tag
docker tag frontend-mvc ahm7dkhalifa/frontend-mvc:v1
docker push ahm7dkhalifa/frontend-mvc:v1
