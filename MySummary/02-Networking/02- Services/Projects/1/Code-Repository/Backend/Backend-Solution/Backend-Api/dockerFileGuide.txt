# Build the image
docker build -t backend-api .

# Run the container
docker run -d -p 8080:8080 --name backend-api backend-api

# Access the API
http://localhost:8080/WeatherForecast

# Publish to a registry (requires login first)
docker tag backend-api yourregistry/backend-api:latest
docker push yourregistry/backend-api:latest

# Example with version tag
docker tag backend-api ahm7dkhalifa/backend-api:v1
docker push ahm7dkhalifa/backend-api:v1
