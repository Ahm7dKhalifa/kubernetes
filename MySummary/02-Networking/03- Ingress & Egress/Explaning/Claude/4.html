⚠ Before — No Ingress
2 dedicated GCP LoadBalancers = $150/month
Plain HTTP — no TLS, credentials in cleartext
Users must know port :38080 or :38282
NodePort open on every node public IP
No NetworkPolicy — any pod reaches any pod
Hardcoded secrets in Deployment YAML
No per-route metrics or rate limiting
✓ After — With Ingress
1 shared GCP LoadBalancer = $75/month
Auto TLS via cert-manager + Let's Encrypt
Clean URLs — /wear and /video, no ports
ClusterIP services only — no NodePort
Calico NetworkPolicy — zero-trust egress
Secrets from K8s Secret refs
Prometheus metrics per route, rate limiting

# WEAR DEPLOYMENT — BEFORE
# :latest tag · hardcoded secrets · no limits
apiVersion: apps/v1
kind: Deployment
metadata:
  name: wear-deployment
  namespace: production
spec:
  replicas: 3
  selector:
    matchLabels: { app: wear }
  template:
    metadata:
      labels: { app: wear }
    spec:
      containers:
      - name: wear
        image: myapp/wear:latest     # ⚠ floating tag
        ports:
        - containerPort: 8080
        env:
        - { name: DB_PASSWORD, value: "wear_pass_123" }  # ⚠ HARDCODED
        - { name: DB_HOST, value: "mysql.production.svc" }
                # ⚠ no resources  → OOM possible
        # ⚠ no securityContext → runs as root
        # ⚠ no readinessProbe → dead pods get traffic
        # ⚠ all 3 pods may land on same node

# INGRESS CONTROLLER — HA 2 replicas, one shared LB
apiVersion: apps/v1
kind: Deployment
metadata:
  name: nginx-ingress-controller
  namespace: ingress-nginx
spec:
  replicas: 2
  selector:
    matchLabels: { app: ingress-nginx }
  template:
    metadata:
      labels: { app: ingress-nginx }
      annotations:
        prometheus.io/scrape: "true"
        prometheus.io/port: "10254"
    spec:
      affinity:
        podAntiAffinity:
          requiredDuringSchedulingIgnoredDuringExecution:
          - labelSelector:
              matchLabels: { app: ingress-nginx }
            topologyKey: kubernetes.io/hostname
      containers:
      - name: controller
        image: registry.k8s.io/ingress-nginx/controller:v1.9.4
        args:
        - /nginx-ingress-controller
        - --publish-service=ingress-nginx/ingress-nginx-svc
        - --ingress-class=nginx
        - --enable-metrics=true
        ports:
        - { name: http,    containerPort: 80    }
        - { name: https,   containerPort: 443   }
        - { name: metrics, containerPort: 10254 }
        resources:
          requests: { cpu: 100m, memory: 90Mi  }
          limits:   { cpu: 500m, memory: 256Mi }
        readinessProbe:
          httpGet: { path: /healthz, port: 10254 }
          periodSeconds: 5
---
# THE ONE SHARED LoadBalancer
apiVersion: v1
kind: Service
metadata:
  name: ingress-nginx-svc
  namespace: ingress-nginx
spec:
  type: LoadBalancer  # ONE for entire cluster
  selector: { app: ingress-nginx }
  ports:
  - { name: http,  port: 80,  targetPort: 80  }
  - { name: https, port: 443, targetPort: 443 }
# INGRESS RESOURCE — path routing, TLS, rate-limit
apiVersion: networking.k8s.io/v1
kind: Ingress
metadata:
  name: store-ingress
  namespace: production
  annotations:
    kubernetes.io/ingress.class: "nginx"
    cert-manager.io/cluster-issuer: "letsencrypt-prod"
    nginx.ingress.kubernetes.io/ssl-redirect: "true"
    nginx.ingress.kubernetes.io/limit-rps: "100"
    nginx.ingress.kubernetes.io/limit-connections: "20"
    nginx.ingress.kubernetes.io/enable-cors: "true"
    nginx.ingress.kubernetes.io/rewrite-target: /$2
    nginx.ingress.kubernetes.io/use-regex: "true"
    nginx.ingress.kubernetes.io/configuration-snippet: |
      more_set_headers "X-Frame-Options: DENY";
      more_set_headers "X-Content-Type-Options: nosniff";
      more_set_headers "Strict-Transport-Security: max-age=31536000";
spec:
  tls:
  - hosts: [www.my-online-store.com]
    secretName: store-tls-secret
  rules:
  - host: www.my-online-store.com
    http:
      paths:
      - path: /wear(/|$)(.*)
        pathType: ImplementationSpecific
        backend:
          service:
            name: wear-service
            port: { number: 80 }
      - path: /video(/|$)(.*)
        pathType: ImplementationSpecific
        backend:
          service:
            name: video-service
            port: { number: 80 }
# CERT-MANAGER — auto TLS, auto-renew
apiVersion: cert-manager.io/v1
kind: ClusterIssuer
metadata:
  name: letsencrypt-prod
spec:
  acme:
    server: https://acme-v02.api.letsencrypt.org/directory
    email: ops@my-online-store.com
    privateKeySecretRef: { name: letsencrypt-prod-key }
    solvers:
    - http01:
        ingress: { class: nginx }
---
apiVersion: cert-manager.io/v1
kind: Certificate
metadata:
  name: store-tls
  namespace: production
spec:
  secretName: store-tls-secret
  duration: 2160h    # 90 days
  renewBefore: 360h  # renew 15d before expiry
  issuerRef:
    name: letsencrypt-prod
    kind: ClusterIssuer

  - www.my-online-store.com
  - wear.my-online-store.com
  - video.my-online-store.com
# WEAR — FULL PRODUCTION DEPLOYMENT
apiVersion: apps/v1
kind: Deployment
metadata:
  name: wear-deployment
  namespace: production
  labels: { app: wear, version: v2.1 }
spec:
  replicas: 3
  selector:
    matchLabels: { app: wear }
  strategy:
    type: RollingUpdate
    rollingUpdate: { maxSurge: 1, maxUnavailable: 0 }
  template:
    metadata:
      labels: { app: wear }
      annotations:
        prometheus.io/scrape: "true"
        prometheus.io/port: "9090"
    spec:
      topologySpreadConstraints:
      - maxSkew: 1
        topologyKey: kubernetes.io/hostname
        whenUnsatisfiable: DoNotSchedule
        labelSelector:
          matchLabels: { app: wear }
      securityContext: { runAsNonRoot: true, runAsUser: 1000 }
      containers:
      - name: wear
        image: myapp/wear:v2.1.0          # pinned
        ports:
        - { name: http,    containerPort: 8080 }
        - { name: metrics, containerPort: 9090 }
        env:
        - name: DB_PASSWORD
          valueFrom:
            secretKeyRef: { name: wear-secrets, key: db-password }
        resources:
          requests: { cpu: 200m, memory: 256Mi  }
          limits:   { cpu: 1000m, memory: 512Mi }
        readinessProbe:
          httpGet: { path: /health, port: 8080 }
          periodSeconds: 5
        livenessProbe:
          httpGet: { path: /health, port: 8080 }
          initialDelaySeconds: 10
        securityContext:
          allowPrivilegeEscalation: false
          readOnlyRootFilesystem: true
          capabilities: { drop: [ALL] }
---

metadata:

apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: allow-ingress-to-wear
  namespace: production
spec:
  podSelector:
    matchLabels: { app: wear }
  policyTypes: [Ingress]
  ingress:
  - from:
    - namespaceSelector:
        matchLabels: { name: ingress-nginx }
      podSelector:
        matchLabels: { app: ingress-nginx }
    ports:
    - { protocol: TCP, port: 8080 }
